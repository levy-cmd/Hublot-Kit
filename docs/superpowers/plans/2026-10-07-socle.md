## Global Constraints

- Module Go : `github.com/levy-cmd/hublot-kit`. Dépendances Go de cette phase : `github.com/moby/moby/client` (et ses dépendances) seulement.
- Routage `net/http` avec motifs Go 1.22+ (`"POST /api/login"`), pas de framework.
- Erreurs API : `{"error": "<code>", "detail": "<texte libre>"}`, codes de la spec §6 uniquement.
- Écoute par défaut `127.0.0.1:7878` ; adresse non locale sans TLS → avertissement au démarrage (§7).
- Jeton : 32 octets `crypto/rand`, base64url sans remplissage, affiché une seule fois ; seule son empreinte SHA-256 (hex) est stockée dans `<data-dir>/token.sha256` (0600) ; comparaison en temps constant.
- Tentatives : 5 échecs par minute et par IP → `429 rate_limited` ; l'IP vient de `RemoteAddr`, jamais de `X-Forwarded-*`.
- Session : 32 octets aléatoires, en mémoire, 12 h ; cookie `hublot_session`, `HttpOnly`, `SameSite=Strict`, `Path=/`, `Secure` si TLS.
- CSRF : toute méthode autre que GET/HEAD exige un `Origin` présent, bien formé, ≠ `null`, égal à l'origine de confiance (`--public-origin`, sinon schéma + `Host`) comparée en entier, sinon `403 forbidden_origin`.
- En-têtes sur toute réponse : `Content-Security-Policy: default-src 'self'; img-src 'self' blob: data:; frame-ancestors 'none'`, `X-Content-Type-Options: nosniff`, `Referrer-Policy: no-referrer`.
- Audit : `<data-dir>/audit.log`, une ligne JSON par événement `{ts, action, container, path, result, error}` + champ `remote` (IP) ajouté par ce plan et reporté dans la spec §7 (tâche 7).
- Interface : `{@html}` interdit (règle ESLint `svelte/no-at-html-tags` en erreur) ; textes dans `web/src/lib/i18n/fr.ts` et `en.ts` avec les mêmes clés ; chaque code d'erreur API a sa traduction.
- Commits en français, terminés par une ligne vide puis `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. **Jeton collé avec espaces ou retour à la ligne** (copier-coller depuis un terminal) : la connexion réussit. → Tâche 7, `TestLoginTrimsToken`.
2. **Docker arrêté ou socket inaccessible** : l'agent démarre quand même ; la liste répond `503 docker_unavailable`, traduit dans l'interface. → Tâche 7, `TestContainersDockerDown` ; tâche 10, clé `error.docker_unavailable`.
3. **Derrière un reverse proxy sans `--public-origin`** : la connexion échoue avec un message clair (`forbidden_origin` traduit), pas une erreur générique. → Tâche 6, `TestOriginMismatchScheme` ; tâche 10, test des clés d'erreur.
4. **Dossier de données impossible à créer** : l'agent s'arrête avec un message qui nomme le dossier, code de sortie 1. → Tâche 9, `TestRunDataDirUnwritable`.
5. **Session expirée en pleine utilisation** : retour à la page de connexion, pas une page vide. → Tâche 11 (garde du layout) et tâche 12 (liste : 401 → `/login`).

```go
// Package config reads the agent's options from flags and HUBLOT_* variables.
package config

import (
	"errors"
	"flag"
	"fmt"
	"io"
	"net/url"
	"strconv"
	"strings"
)

type Config struct {
	Listen       string
	DataDir      string
	ReadOnly     bool
	TLSCert      string
	TLSKey       string
	PublicOrigin string
}

// TLS reports whether the agent serves HTTPS itself.
func (c Config) TLS() bool { return c.TLSCert != "" }

// Load parses args. Precedence: flag, then HUBLOT_* variable, then default.
func Load(args []string, getenv func(string) string) (Config, error) {
	or := func(name, def string) string {
		if v := getenv(name); v != "" {
			return v
		}
		return def
	}
	readOnly, err := strconv.ParseBool(or("HUBLOT_READ_ONLY", "false"))
	if err != nil {
		return Config{}, fmt.Errorf("HUBLOT_READ_ONLY: %w", err)
	}

	var c Config
	fs := flag.NewFlagSet("hublot", flag.ContinueOnError)
	fs.SetOutput(io.Discard)
	fs.StringVar(&c.Listen, "listen", or("HUBLOT_LISTEN", "127.0.0.1:7878"), "adresse d'écoute")
	fs.StringVar(&c.DataDir, "data-dir", or("HUBLOT_DATA_DIR", "/var/lib/hublot-kit"), "dossier de données")
	fs.BoolVar(&c.ReadOnly, "read-only", readOnly, "désactive toute écriture")
	fs.StringVar(&c.TLSCert, "tls-cert", or("HUBLOT_TLS_CERT", ""), "certificat TLS")
	fs.StringVar(&c.TLSKey, "tls-key", or("HUBLOT_TLS_KEY", ""), "clé TLS")
	fs.StringVar(&c.PublicOrigin, "public-origin", or("HUBLOT_PUBLIC_ORIGIN", ""), "origine publique derrière un reverse proxy")
	if err := fs.Parse(args); err != nil {
		return Config{}, err
	}

	if (c.TLSCert == "") != (c.TLSKey == "") {
		return Config{}, errors.New("--tls-cert et --tls-key vont ensemble")
	}
	if c.PublicOrigin != "" {
		o, err := normalizeOrigin(c.PublicOrigin)
		if err != nil {
			return Config{}, fmt.Errorf("--public-origin: %w", err)
		}
		c.PublicOrigin = o
	}
	return c, nil
}

// normalizeOrigin returns scheme://host[:port] in lower case, or an error.
func normalizeOrigin(s string) (string, error) {
	u, err := url.Parse(s)
	if err != nil {
		return "", err
	}
	scheme := strings.ToLower(u.Scheme)
	if scheme != "http" && scheme != "https" {
		return "", errors.New("schéma http ou https attendu")
	}
	if u.Host == "" {
		return "", errors.New("hôte manquant")
	}
	if u.Path != "" && u.Path != "/" || u.RawQuery != "" || u.Fragment != "" {
		return "", errors.New("une origine n'a ni chemin, ni requête, ni fragment")
	}
	return scheme + "://" + strings.ToLower(u.Host), nil
}
```

```go
// Package audit writes the append-only JSON-lines audit log.
package audit

import (
	"encoding/json"
	"os"
	"sync"
	"time"
)

type Event struct {
	TS        time.Time `json:"ts"`
	Action    string    `json:"action"`
	Container string    `json:"container,omitempty"`
	Path      string    `json:"path,omitempty"`
	Result    string    `json:"result"`
	Error     string    `json:"error,omitempty"`
	Remote    string    `json:"remote,omitempty"`
}

type Log struct {
	mu sync.Mutex
	f  *os.File
}

func Open(path string) (*Log, error) {
	f, err := os.OpenFile(path, os.O_WRONLY|os.O_APPEND|os.O_CREATE, 0o600)
	if err != nil {
		return nil, err
	}
	return &Log{f: f}, nil
}

// Write appends one event as a single JSON line. TS defaults to now (UTC).
func (l *Log) Write(e Event) error {
	if e.TS.IsZero() {
		e.TS = time.Now().UTC()
	}
	b, err := json.Marshal(e)
	if err != nil {
		return err
	}
	b = append(b, '\n')
	l.mu.Lock()
	defer l.mu.Unlock()
	_, err = l.f.Write(b)
	return err
}

func (l *Log) Close() error { return l.f.Close() }
```

```go
// Package auth holds the access token, sessions and the login rate limiter.
package auth

import (
	"crypto/rand"
	"crypto/sha256"
	"crypto/subtle"
	"encoding/base64"
	"encoding/hex"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"strings"
)

const tokenFile = "token.sha256"

// GenerateToken returns 32 random bytes as unpadded base64url (43 characters).
func GenerateToken() (string, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b), nil
}

func digest(token string) []byte {
	s := sha256.Sum256([]byte(token))
	return s[:]
}

// EnsureToken creates the token on first launch. The plain token is returned
// only when it was just created; it is never stored.
func EnsureToken(dataDir string) (plain string, created bool, err error) {
	_, err = os.Stat(filepath.Join(dataDir, tokenFile))
	if err == nil {
		return "", false, nil
	}
	if !errors.Is(err, fs.ErrNotExist) {
		return "", false, err
	}
	plain, err = ResetToken(dataDir)
	return plain, err == nil, err
}

// ResetToken generates a new token and replaces the stored digest atomically.
func ResetToken(dataDir string) (string, error) {
	tok, err := GenerateToken()
	if err != nil {
		return "", err
	}
	tmp, err := os.CreateTemp(dataDir, ".token-*")
	if err != nil {
		return "", err
	}
	defer os.Remove(tmp.Name())
	if err := tmp.Chmod(0o600); err != nil {
		tmp.Close()
		return "", err
	}
	if _, err := tmp.WriteString(hex.EncodeToString(digest(tok)) + "\n"); err != nil {
		tmp.Close()
		return "", err
	}
	if err := tmp.Close(); err != nil {
		return "", err
	}
	if err := os.Rename(tmp.Name(), filepath.Join(dataDir, tokenFile)); err != nil {
		return "", err
	}
	return tok, nil
}

type Verifier struct{ digest []byte }

func LoadVerifier(dataDir string) (*Verifier, error) {
	raw, err := os.ReadFile(filepath.Join(dataDir, tokenFile))
	if err != nil {
		return nil, err
	}
	d, err := hex.DecodeString(strings.TrimSpace(string(raw)))
	if err != nil || len(d) != sha256.Size {
		return nil, fmt.Errorf("%s corrompu : relancez `hublot reset-token`", tokenFile)
	}
	return &Verifier{digest: d}, nil
}

// Check compares digests in constant time.
func (v *Verifier) Check(candidate string) bool {
	return subtle.ConstantTimeCompare(digest(candidate), v.digest) == 1
}
```

```go
package auth

import (
	"sync"
	"time"
)

const SessionTTL = 12 * time.Hour

type Sessions struct {
	mu      sync.Mutex
	ttl     time.Duration
	now     func() time.Time
	expires map[string]time.Time
}

func NewSessions(ttl time.Duration, now func() time.Time) *Sessions {
	return &Sessions{ttl: ttl, now: now, expires: map[string]time.Time{}}
}

func (s *Sessions) Create() (string, error) {
	id, err := GenerateToken()
	if err != nil {
		return "", err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	now := s.now()
	for k, exp := range s.expires { // purge expired sessions
		if !now.Before(exp) {
			delete(s.expires, k)
		}
	}
	s.expires[id] = now.Add(s.ttl)
	return id, nil
}

func (s *Sessions) Valid(id string) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	exp, ok := s.expires[id]
	if !ok {
		return false
	}
	if !s.now().Before(exp) {
		delete(s.expires, id)
		return false
	}
	return true
}

func (s *Sessions) Delete(id string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	delete(s.expires, id)
}
```

```go
package auth

import (
	"sync"
	"time"
)

// Limiter counts failures per key over a sliding window.
type Limiter struct {
	mu     sync.Mutex
	max    int
	window time.Duration
	now    func() time.Time
	fails  map[string][]time.Time
}

func NewLimiter(max int, window time.Duration, now func() time.Time) *Limiter {
	return &Limiter{max: max, window: window, now: now, fails: map[string][]time.Time{}}
}

func (l *Limiter) prune(key string) []time.Time {
	cutoff := l.now().Add(-l.window)
	kept := l.fails[key][:0]
	for _, t := range l.fails[key] {
		if t.After(cutoff) {
			kept = append(kept, t)
		}
	}
	if len(kept) == 0 {
		delete(l.fails, key)
		return nil
	}
	l.fails[key] = kept
	return kept
}

func (l *Limiter) Allowed(key string) bool {
	l.mu.Lock()
	defer l.mu.Unlock()
	return len(l.prune(key)) < l.max
}

func (l *Limiter) Fail(key string) {
	l.mu.Lock()
	defer l.mu.Unlock()
	l.fails[key] = append(l.prune(key), l.now())
}
```

```go
// Package dockerx is the agent's only gateway to the Docker Engine API.
package dockerx

import (
	"context"
	"sort"
	"strings"
	"time"

	"github.com/moby/moby/api/types/container"
	"github.com/moby/moby/client"
)

type Port struct {
	Private uint16 `json:"private"`
	Public  uint16 `json:"public,omitempty"`
	Type    string `json:"type"`
	IP      string `json:"ip,omitempty"`
}

type Container struct {
	ID      string    `json:"id"`
	Name    string    `json:"name"`
	Image   string    `json:"image"`
	State   string    `json:"state"`
	Status  string    `json:"status"`
	Created time.Time `json:"created"`
	Ports   []Port    `json:"ports"`
}

type Client interface {
	ListContainers(ctx context.Context) ([]Container, error)
	Ping(ctx context.Context) error
}

type engine struct{ c *client.Client }

// New connects lazily: no request is sent until the first call.
// The API version is negotiated on the first request.
func New() (Client, error) {
	c, err := client.New(client.FromEnv)
	if err != nil {
		return nil, err
	}
	return &engine{c: c}, nil
}

func (e *engine) Ping(ctx context.Context) error {
	_, err := e.c.Ping(ctx, client.PingOptions{})
	return err
}

func (e *engine) ListContainers(ctx context.Context) ([]Container, error) {
	res, err := e.c.ContainerList(ctx, client.ContainerListOptions{All: true})
	if err != nil {
		return nil, err
	}
	out := make([]Container, 0, len(res.Items))
	for _, s := range res.Items {
		out = append(out, fromSummary(s))
	}
	sort.Slice(out, func(i, j int) bool { return out[i].Name < out[j].Name })
	return out, nil
}

func fromSummary(s container.Summary) Container {
	name := ""
	if len(s.Names) > 0 {
		name = strings.TrimPrefix(s.Names[0], "/")
	}
	if name == "" {
		name = s.ID[:min(12, len(s.ID))]
	}
	type portKey struct {
		private, public uint16
		typ             string
	}
	ports := make([]Port, 0, len(s.Ports))
	seen := map[portKey]bool{}
	for _, p := range s.Ports {
		key := portKey{p.PrivatePort, p.PublicPort, p.Type}
		if seen[key] {
			continue // same mapping on IPv4 and IPv6
		}
		seen[key] = true
		ip := ""
		if p.IP.IsValid() {
			ip = p.IP.String()
		}
		ports = append(ports, Port{Private: p.PrivatePort, Public: p.PublicPort, Type: p.Type, IP: ip})
	}
	return Container{
		ID:      s.ID,
		Name:    name,
		Image:   s.Image,
		State:   string(s.State),
		Status:  s.Status,
		Created: time.Unix(s.Created, 0).UTC(),
		Ports:   ports,
	}
}
```

```go
package api

import (
	"net"
	"net/http"
	"net/url"
	"strings"

	"github.com/levy-cmd/hublot-kit/internal/auth"
)

const cookieName = "hublot_session"

func securityHeaders(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		h := w.Header()
		h.Set("Content-Security-Policy", "default-src 'self'; img-src 'self' blob: data:; frame-ancestors 'none'")
		h.Set("X-Content-Type-Options", "nosniff")
		h.Set("Referrer-Policy", "no-referrer")
		next.ServeHTTP(w, r)
	})
}

// originChecker enforces the CSRF rule of spec §7 on every non-GET/HEAD request.
type originChecker struct {
	public string // normalized --public-origin, or ""
	tls    bool
}

func (o originChecker) trusted(r *http.Request) string {
	if o.public != "" {
		return o.public
	}
	scheme := "http"
	if o.tls {
		scheme = "https"
	}
	return scheme + "://" + strings.ToLower(r.Host)
}

func (o originChecker) wrap(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method == http.MethodGet || r.Method == http.MethodHead {
			next.ServeHTTP(w, r)
			return
		}
		origin := r.Header.Get("Origin")
		u, err := url.Parse(origin)
		if origin == "" || origin == "null" || err != nil || u.Scheme == "" || u.Host == "" || u.Path != "" {
			writeError(w, http.StatusForbidden, "forbidden_origin", "missing or malformed Origin")
			return
		}
		if strings.ToLower(u.Scheme)+"://"+strings.ToLower(u.Host) != o.trusted(r) {
			writeError(w, http.StatusForbidden, "forbidden_origin", "Origin "+origin+" is not "+o.trusted(r))
			return
		}
		next.ServeHTTP(w, r)
	})
}

func requireSession(s *auth.Sessions, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		c, err := r.Cookie(cookieName)
		if err != nil || !s.Valid(c.Value) {
			writeError(w, http.StatusUnauthorized, "unauthorized", "")
			return
		}
		next.ServeHTTP(w, r)
	})
}

// clientIP uses the TCP peer only; X-Forwarded-* headers are never trusted.
func clientIP(r *http.Request) string {
	host, _, err := net.SplitHostPort(r.RemoteAddr)
	if err != nil {
		return r.RemoteAddr
	}
	return host
}
```

```go
package api

import (
	"encoding/json"
	"io"
	"log/slog"
	"net/http"
	"strings"

	"github.com/levy-cmd/hublot-kit/internal/audit"
	"github.com/levy-cmd/hublot-kit/internal/auth"
	"github.com/levy-cmd/hublot-kit/internal/dockerx"
)

type Deps struct {
	Docker       dockerx.Client
	Verifier     *auth.Verifier
	Sessions     *auth.Sessions
	Limiter      *auth.Limiter
	Audit        *audit.Log
	Version      string
	ReadOnly     bool
	TLS          bool
	PublicOrigin string       // normalized, or ""
	Static       http.Handler // embedded UI; nil in API tests
}

type server struct{ d Deps }

func New(d Deps) http.Handler {
	s := &server{d: d}
	session := func(h http.HandlerFunc) http.Handler { return requireSession(d.Sessions, h) }

	mux := http.NewServeMux()
	mux.HandleFunc("POST /api/login", s.login)
	mux.Handle("POST /api/logout", session(s.logout))
	mux.Handle("GET /api/meta", session(s.meta))
	mux.Handle("GET /api/containers", session(s.containers))
	mux.HandleFunc("/api/", func(w http.ResponseWriter, r *http.Request) {
		writeError(w, http.StatusNotFound, "not_found", "")
	})
	if d.Static != nil {
		mux.Handle("/", d.Static)
	}
	return securityHeaders(originChecker{public: d.PublicOrigin, tls: d.TLS}.wrap(mux))
}

func (s *server) login(w http.ResponseWriter, r *http.Request) {
	ip := clientIP(r)
	if !s.d.Limiter.Allowed(ip) {
		writeError(w, http.StatusTooManyRequests, "rate_limited", "")
		return
	}
	var body struct {
		Token string `json:"token"`
	}
	err := json.NewDecoder(io.LimitReader(r.Body, 4096)).Decode(&body)
	if err != nil || !s.d.Verifier.Check(strings.TrimSpace(body.Token)) {
		s.d.Limiter.Fail(ip)
		if err := s.d.Audit.Write(audit.Event{Action: "login.fail", Result: "refused", Remote: ip}); err != nil {
			slog.Error("audit write", "err", err)
		}
		writeError(w, http.StatusUnauthorized, "unauthorized", "")
		return
	}
	id, err := s.d.Sessions.Create()
	if err != nil {
		writeError(w, http.StatusInternalServerError, "internal", err.Error())
		return
	}
	http.SetCookie(w, &http.Cookie{
		Name: cookieName, Value: id, Path: "/",
		HttpOnly: true, SameSite: http.SameSiteStrictMode, Secure: s.d.TLS,
		MaxAge: int(auth.SessionTTL.Seconds()),
	})
	writeJSON(w, http.StatusOK, map[string]bool{"ok": true})
}

func (s *server) logout(w http.ResponseWriter, r *http.Request) {
	if c, err := r.Cookie(cookieName); err == nil {
		s.d.Sessions.Delete(c.Value)
	}
	http.SetCookie(w, &http.Cookie{Name: cookieName, Value: "", Path: "/", HttpOnly: true, SameSite: http.SameSiteStrictMode, Secure: s.d.TLS, MaxAge: -1})
	w.WriteHeader(http.StatusNoContent)
}

func (s *server) meta(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{"version": s.d.Version, "readOnly": s.d.ReadOnly})
}

func (s *server) containers(w http.ResponseWriter, r *http.Request) {
	list, err := s.d.Docker.ListContainers(r.Context())
	if err != nil {
		writeError(w, http.StatusServiceUnavailable, "docker_unavailable", err.Error())
		return
	}
	if list == nil {
		list = []dockerx.Container{}
	}
	writeJSON(w, http.StatusOK, list)
}
```

```go
// Package web serves the compiled SvelteKit UI embedded in the binary.
package web

import (
	"embed"
	"io/fs"
	"net/http"
	"path"
	"strings"
)

//go:embed all:dist
var dist embed.FS

func Handler() http.Handler {
	sub, err := fs.Sub(dist, "dist")
	if err != nil {
		panic(err) // "dist" is always embedded
	}
	return NewHandler(sub)
}

func NewHandler(files fs.FS) http.Handler {
	fileServer := http.FileServerFS(files)
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		p := strings.TrimPrefix(path.Clean(r.URL.Path), "/")
		if p != "" {
			if st, err := fs.Stat(files, p); err == nil && !st.IsDir() {
				fileServer.ServeHTTP(w, r)
				return
			}
		}
		index, err := fs.ReadFile(files, "index.html")
		if err != nil {
			http.Error(w, "interface non compilée : lancez `make web`", http.StatusServiceUnavailable)
			return
		}
		w.Header().Set("Content-Type", "text/html; charset=utf-8")
		w.Write(index)
	})
}
```

```go
	if err != nil {
		fmt.Fprintln(stderr, "journal d'audit :", err)

	if err != nil {
		fmt.Fprintln(stderr, "client Docker :", err)
		return 1

		fmt.Fprintf(stderr, "ATTENTION : écoute sur %s sans TLS. À réserver à un conteneur publié sur 127.0.0.1 ou à un reverse proxy HTTPS.\n", cfg.Listen)
	}

	handler := api.New(api.Deps{
		Docker: docker, Verifier: verifier, Audit: auditLog, Version: version,
		Sessions: auth.NewSessions(auth.SessionTTL, time.Now),
		Limiter:  auth.NewLimiter(5, time.Minute, time.Now),
		ReadOnly: cfg.ReadOnly, TLS: cfg.TLS(), PublicOrigin: cfg.PublicOrigin,
		Static: web.Handler(),

		return 1
```

```ts
export type { Key };
export type Lang = 'fr' | 'en';

// Error codes of spec §6, plus client-side "network" and "unknown".
export const API_ERROR_CODES = [
	'unauthorized', 'invalid_name', 'forbidden_origin', 'rate_limited', 'not_found', 'read_only',
	'conflict', 'container_changed', 'indexing', 'too_large', 'not_regular', 'forbidden_path',
	'link_in_path', 'docker_unavailable', 'internal', 'network', 'unknown'
] as const;

const dicts: Record<Lang, Record<Key, string>> = { fr, en };

export function detectLang(navLang: string | undefined): Lang {
	return navLang?.toLowerCase().startsWith('fr') ? 'fr' : 'en';
}

const lang: Lang = typeof navigator === 'undefined' ? 'fr' : detectLang(navigator.language);

export function t(key: Key): string {
	return dicts[lang][key];
}

export function errorMessage(code: string): string {
	const key = `error.${code}`;
	return key in fr ? t(key as Key) : t('error.unknown');
}
```

```ts
export class ApiError extends Error {
	constructor(
		public status: number,
		public code: string,
		public detail?: string
	) {
		super(code);
	}
}

export async function api<T>(path: string, init: RequestInit = {}): Promise<T> {
	let res: Response;
	try {
		res = await fetch(`/api${path}`, {
			credentials: 'same-origin',
			...init,
			headers: { 'Content-Type': 'application/json', ...(init.headers ?? {}) }
		});
	} catch {
		throw new ApiError(0, 'network');
	}
	if (res.status === 204) return undefined as T;
	const body = await res.json().catch(() => null);
	if (!res.ok) throw new ApiError(res.status, body?.error ?? 'unknown', body?.detail);
	return body as T;
}

export type Port = { private: number; public?: number; type: string; ip?: string };
export type Container = {
	id: string;
	name: string;
	image: string;
	state: string;
	status: string;
	created: string;
	ports: Port[];
};
export type Meta = { version: string; readOnly: boolean };

export const login = (token: string) =>
	api<{ ok: boolean }>('/login', { method: 'POST', body: JSON.stringify({ token }) });
export const logout = () => api<void>('/logout', { method: 'POST' });
export const getMeta = () => api<Meta>('/meta');
export const listContainers = () => api<Container[]>('/containers');
```

```ts
import type { Container, Port } from './api';
import { t, type Key } from './i18n';

export function formatPorts(ports: Port[]): string {
	return [...ports]
		.sort((a, b) => (a.public ?? a.private) - (b.public ?? b.private))
		.map((p) => (p.public ? `${p.public}→${p.private}/${p.type}` : `${p.private}/${p.type}`))
		.join(', ');
}

export function filterContainers(list: Container[], query: string, showAll: boolean): Container[] {
	const q = query.trim().toLowerCase();
	return list.filter(
		(c) =>
			(showAll || c.state === 'running') &&
			(q === '' || c.name.toLowerCase().includes(q) || c.image.toLowerCase().includes(q))
	);
}

const STATES = ['running', 'exited', 'created', 'paused', 'restarting', 'removing', 'dead'];

export function stateLabel(state: string): string {
	return STATES.includes(state) ? t(`state.${state}` as Key) : state;
}
```
